٤. مخطط قاعدة البيانات المقترح (Supabase)

4 جداول جديدة، بادئة trader_ لتفادي أي تصادم مع جداول bare-academy الحالية (students, programs, users...). منسوخ حرفياً من Section G في ملف التعليمات، مع RLS مفعّل لكل جدول (حتى لو بسياسة متساهلة):

-- 1) المستخدمون (مقيّمون + مدير)
CREATE TABLE trader_users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  username TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,              -- نص صريح، بقرار المالك (انظر القرار الأمني ج)
  full_name TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('admin', 'evaluator')),
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2) معايير التقييم (ديناميكية، المدير يضيف/يعدّل)
CREATE TABLE trader_criteria (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  max_score INT NOT NULL CHECK (max_score BETWEEN 1 AND 10),
  sort_order INT NOT NULL DEFAULT 1,
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 3) المتاجر
CREATE TABLE trader_stores (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  first_name TEXT NOT NULL,
  last_name TEXT NOT NULL,
  store_type TEXT,
  created_by TEXT NOT NULL REFERENCES trader_users(username) ON UPDATE CASCADE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 4) التقييمات
CREATE TABLE trader_evaluations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  store_id UUID NOT NULL REFERENCES trader_stores(id) ON DELETE CASCADE,
  evaluator_username TEXT NOT NULL REFERENCES trader_users(username) ON UPDATE CASCADE,
  scores JSONB NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (store_id, evaluator_username)
);

CREATE INDEX trader_eval_store_idx ON trader_evaluations(store_id);
CREATE INDEX trader_eval_user_idx ON trader_evaluations(evaluator_username);
CREATE INDEX trader_criteria_active_idx ON trader_criteria(is_active, sort_order);

CREATE OR REPLACE FUNCTION trader_touch_updated_at()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = NOW(); RETURN NEW; END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trader_evaluations_touch
BEFORE UPDATE ON trader_evaluations
FOR EACH ROW EXECUTE FUNCTION trader_touch_updated_at();

-- Row Level Security
ALTER TABLE trader_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE trader_criteria ENABLE ROW LEVEL SECURITY;
ALTER TABLE trader_stores ENABLE ROW LEVEL SECURITY;
ALTER TABLE trader_evaluations ENABLE ROW LEVEL SECURITY;

-- سياسات متساهلة (مقايضة مقصودة لفعالية داخلية — انظر القرار ج)
CREATE POLICY "trader_users_anon_all" ON trader_users FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "trader_criteria_anon_all" ON trader_criteria FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "trader_stores_anon_all" ON trader_stores FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "trader_evaluations_anon_all" ON trader_evaluations FOR ALL USING (true) WITH CHECK (true);

ماذا تخسر لو غيّرت رأيك لاحقاً: لو كبر هذا المنتج (فعاليات متكررة، مستخدمون أكثر) وأردنا فصله لاحقاً، النقل ممكن لأن الملفات مستقلة في مجلدين (trader/, view/) — يتطلب ساعات عمل لا دقائق، لكنه ليس مستحيلاً.

٩. خطوات تنفيذ Phase 1 (بعد موافقتك فقط — لن تبدأ قبلها)
